**优化建议**：
```typescript
function inferDifficulty(content: string, analysis: string): 'easy' | 'medium' | 'hard' {
  const text = content + analysis
  const len = text.length

  if (len > 300) return 'hard'
  if (len < 80) return 'easy'

  if (text.includes('综合') || text.includes('归纳') || text.includes('概括') || text.includes('分析')) return 'hard'
  if (text.includes('直接') || text.includes('属于') || text.includes('标志') || text.includes('是')) return 'easy'

  return 'medium'
}
```
